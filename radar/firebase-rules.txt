REGRAS FIRESTORE

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function ownProfile(userId) {
      return signedIn() && request.auth.uid == userId;
    }

    function isApproved() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.status == "approved";
    }

    function isAdmin() {
      return isApproved()
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    match /resources/{resourceId} {
      allow read: if resource.data.status == "published" || isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow create: if ownProfile(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.status == "pending"
        && request.resource.data.role == "member";

      allow read: if ownProfile(userId) || isAdmin();
      allow update: if isAdmin();
      allow delete: if isAdmin() && resource.data.status == "pending";
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}


REGRAS STORAGE

rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {
    function isAdminUid() {
      return request.auth != null
        && request.auth.uid in [
          "COLE_SEU_UID_ADMIN_AQUI"
        ];
    }

    match /resources/{resourceId}/{fileName} {
      allow read: if true;
      allow create, update: if isAdminUid()
        && request.resource.size < 1 * 1024 * 1024
        && request.resource.contentType == "image/webp";
      allow delete: if isAdminUid();
    }

    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
